def normalize_text(text):
    """
    Normalize Unicode characters and replace special characters.
    """
    text = unicodedata.normalize('NFKC', text)
    # Replace specific ligature characters
    ligatures = {
        'ﬁ': 'fi',
        'ﬂ': 'fl',
        'ﬀ': 'ff',
        'ﬃ': 'ffi',
        'ﬄ': 'ffl',
        'ﬅ': 'st',
        'ﬆ': 'st',
        '\ufb01': 'fi',
        '\ufb02': 'fl',
    }
    for ligature, replacement in ligatures.items():
        text = text.replace(ligature, replacement)
    # Replace curly quotes and other special characters
    curly_quotes = {
        '“': '"',
        '”': '"',
        '‘': "'",
        '’': "'",   # This replaces curly apostrophe with straight apostrophe
        '–': '-',   # En dash
        '—': '-',   # Em dash
        '…': '...', # Ellipsis
        '′': "'",   # Prime
        '″': '"',   # Double Prime
    }
    for curly, straight in curly_quotes.items():
        text = text.replace(curly, straight)
    return text

def prepare_text_for_matching(text):
    """
    Normalize text by replacing special characters, removing extra spaces and line breaks.
    """
    text = normalize_text(text)
    # Replace multiple spaces and line breaks with a single space
    text = re.sub(r'\s+', ' ', text)    
    #text = text.replace('\n', ' ')
    #text = ' '.join(text.split())  # Remove extra spaces
    return text.lower()

def remove_punctuation(txt):
    """
    Remove punctuation from a given string.
    """
    return txt.translate(str.maketrans('', '', string.punctuation))

def is_poetry(text):
    """
    Simple heuristic to determine if the text is poetry based on line length variability and average line length.
    """
    lines = text.strip().split('\n')
    if len(lines) < 3:
        return False
    line_lengths = [len(line.strip()) for line in lines]
    average_length = sum(line_lengths) / len(line_lengths)
    # If lines are relatively short and length varies less, it's likely poetry
    if average_length < 60 and max(line_lengths) - min(line_lengths) < 20:
        return True
    return False


def highlight_query(text, query):
    """
    Highlights exact matches of the query in the text.
    """
    # Normalize text and query
    normalized_text = prepare_text_for_matching(text)
    normalized_query = prepare_text_for_matching(query)
    
    # Escape special characters in the query and add word boundaries
    query_regex = r'\b' + re.escape(normalized_query) + r'\b'
    
    # Find all matches in the normalized text
    matches = list(re.finditer(query_regex, normalized_text, flags=re.IGNORECASE))
    
    # Offset to adjust positions due to added <mark> tags
    offset = 0
    original_text = text
    
    for match in matches:
        start, end = match.span()
        # Adjust positions based on previous replacements
        start += offset
        end += offset
        
        # Extract the original text corresponding to the matched span
        original_text_span = original_text[start:end]
        
        # Replace the original text with highlighted text
        highlighted_span = f"<mark>{original_text_span}</mark>"
        
        # Update the text with the highlighted span
        original_text = original_text[:start] + highlighted_span + original_text[end:]
        
        # Update the offset
        offset += len(highlighted_span) - (end - start)
    
    return original_text

def extract_matching_sentences_new(text, query, max_lines=10, min_chars=200, max_chars=500):
    lines = text.split('\n')
    normalized_query = prepare_text_for_matching(query)
    found_indices = []
    
    for i, line in enumerate(lines):
        line_normalized = prepare_text_for_matching(line)
        if normalized_query in line_normalized:
            found_indices.append(i)
        elif any(word in line_normalized for word in normalized_query.split()):
            found_indices.append(i)
                
    if found_indices:
        start = max(0, found_indices[0] - 5)
        end = min(len(lines), found_indices[-1] + 6)
        context_lines = lines[start:end]
        snippet = '\n'.join(context_lines)
        snippet = highlight_query(snippet, query)
        
        # Ensure snippet meets min_chars requirement
        if len(snippet) < min_chars:
            start = max(0, start - 5)
            end = min(len(lines), end + 5)
            context_lines = lines[start:end]
            snippet = '\n'.join(context_lines)
            snippet = highlight_query(snippet, query)
    else:
        snippet = text[:max_chars]
        snippet = highlight_query(snippet, query)
    
    # Handle line breaks based on content type
    if is_poetry(snippet):
        # For poetry, replace single line breaks with <br>
        snippet = snippet.replace('\n', '<br>')
    else:
        # For prose, replace line breaks within paragraphs with spaces
        snippet = snippet.replace('\n', ' ')
    
    # Replace multiple spaces with a single space
    snippet = re.sub(r'\s+', ' ', snippet)
    
    # Truncate to max_chars
    snippet = snippet[:max_chars]
    return snippet.strip()

def extract_matching_sentences(text, query, max_chars=500):
    logging.info("Entering extract_matching_sentences")
    logging.info(f"Initial text length: {len(text)}")
    logging.info(f"Query: {query}")

    # Remove existing HTML tags like <br> and <mark> to avoid confusion in normalization
    clean_text = bleach.clean(text, tags=[], strip=True)
    logging.info(f"Text after removing HTML tags length: {len(clean_text)}")

    def remove_punctuation(txt):
        return txt.translate(str.maketrans('', '', string.punctuation))

    # Create a single-line version (just in case)
    single_line_text = clean_text.replace('\n', ' ')
    logging.info(f"Single line text length: {len(single_line_text)}")

    # Normalize and remove punctuation
    normalized_query = remove_punctuation(prepare_text_for_matching(query))
    normalized_text = remove_punctuation(prepare_text_for_matching(single_line_text))

    logging.info(f"Normalized query (no punctuation): {normalized_query}")
    logging.info(f"Normalized text (no punctuation) length: {len(normalized_text)}")

    start_index = normalized_text.find(normalized_query)
    logging.info(f"Match start_index: {start_index}")

    if start_index == -1:
        # No match found: just return a snippet from the original text
        snippet = text[:max_chars]
        snippet = highlight_query(snippet, query)
    else:
        # Provide context around the found match
        snippet_start = max(0, start_index - 200)
        snippet_end = min(len(single_line_text), start_index + len(normalized_query) + 300)
        logging.info(f"Snippet start: {snippet_start}, Snippet end: {snippet_end}")

        # Extract snippet from the original text based on these indices
        # Because the only difference between single_line_text and clean_text is newlines replaced with space,
        # and we removed HTML tags, lengths should be close. 
        # We'll do a best-effort approach here: since indexing might be slightly off if original had multiple spaces,
        # we focus on substring length. This is a heuristic approach.

        # We'll attempt to find a substring in the original text that roughly corresponds to snippet_start/snippet_end:
        # A simpler approach: just slice the original text by character indices, trusting indexing isn't too off.
        # If needed, adjust this logic to map indexes more precisely.
        snippet = text
        # Let's try to map the snippet range in terms of the cleaned text length vs original
        # Since we only replaced newlines with spaces when creating single_line_text, 
        # their lengths should be identical except for HTML tags removed earlier.
        # We'll just use snippet_start/snippet_end on clean_text length, but we no longer have line breaks.
        # Let's just trust snippet_start/snippet_end as they are from the normalized text.
        
        # This is a rough heuristic. Ideally, we track original and normalized offsets more precisely.
        snippet = text[:max_chars + snippet_start] if snippet_start > 0 else text
        if snippet_end < len(text):
            snippet = text[snippet_start:snippet_end]

        logging.debug(f"Snippet length before highlight: {len(snippet)}")
        snippet = highlight_query(snippet, query)
        logging.debug(f"Snippet length after highlight: {len(snippet)}")

    # Handle line breaks
    if is_poetry(snippet):
        snippet = snippet.replace('\n', '<br>')
    else:
        snippet = snippet.replace('\n', ' ')

    # Replace multiple spaces
    snippet = re.sub(r'\s+', ' ', snippet)
    # Allow only <br> and <mark> tags
    allowed_tags = ['br', 'mark']
    snippet = bleach.clean(snippet, tags=allowed_tags, strip=True)
    snippet = snippet[:max_chars].strip()

    logging.debug(f"Final snippet length: {len(snippet)}")
    logging.debug("Exiting extract_matching_sentences")
    return snippet


def extract_matching_sentences_last(text, query, max_lines=10, min_chars=200, max_chars=500):
    lines = text.split('\n')
    normalized_query = prepare_text_for_matching(query)
    found_indices = []
    
    # Modified search logic
    for i, line in enumerate(lines):
        line_normalized = prepare_text_for_matching(line)
        # Check if the entire normalized query appears in the normalized line
        if normalized_query in line_normalized:
            found_indices.append(i)
        # As a fallback, check for individual words
        elif any(word in line_normalized for word in normalized_query.split()):
            found_indices.append(i)
            
    if found_indices:
        start = max(0, found_indices[0] - 5)
        end = min(len(lines), found_indices[-1] + 6)
        context_lines = lines[start:end]
        highlighted_lines = [highlight_query(line, query) for line in context_lines]
        snippet = '\n'.join(highlighted_lines)
        
        # Ensure snippet meets min_chars requirement
        if len(snippet) < min_chars:
            start = max(0, start - 5)
            end = min(len(lines), end + 5)
            context_lines = lines[start:end]
            highlighted_lines = [highlight_query(line, query) for line in context_lines]
            snippet = '\n'.join(highlighted_lines)
    else:
        # If no matching line is found, return the beginning of the text
        snippet = text[:max_chars]
        snippet = highlight_query(snippet, query)
        
    # Replace newlines with '<br/>'
    snippet = snippet.replace('\n', '<br/>')
    # Allow only <br> and <mark> tags
    allowed_tags = ['br', 'mark']
    snippet = bleach.clean(snippet, tags=allowed_tags, strip=True)
    # Truncate to max_chars
    snippet = snippet[:max_chars]
    return snippet


def apply_filters(results, filters):
    filtered = []
    for result in results:
        match = True
        for key, value in filters.items():
            if key not in result or result[key] != value:
                match = False
                break
        if match:
            filtered.append(result)
    return filtered
